function MenuItem({ icon, label, active, badge, to }: { icon: string; label: string; active?: boolean; badge?: string; to?: string }) {
  const content = (
    <div style={{
      display: 'flex',
      alignItems: 'center',
      gap: '12px',
      padding: '8px 12px',
      borderRadius: '6px',
      background: active ? '#ecf0f1' : 'transparent',
      cursor: 'pointer',
      position: 'relative',
      textDecoration: 'none',
      color: 'inherit'
    }}>
      <span style={{ fontSize: '14px' }}>{icon}</span>
      <span style={{ 
        fontSize: '13px', 
        color: active ? '#2c3e50' : '#7f8c8d',
        fontWeight: active ? '600' : '400'
      }}>{label}</span>
      {badge && (
        <span style={{
          position: 'absolute',
          right: '12px',
          background: '#e74c3c',
          color: '#ffffff',
          fontSize: '10px',
          fontWeight: '600',
          padding: '2px 6px',
          borderRadius: '8px',
          minWidth: '14px',
          textAlign: 'center'
        }}>{badge}</span>
      )}
    </div>
  );

  return to ? <Link to={to} style={{ textDecoration: 'none', color: 'inherit' }}>{content}</Link> : content;
}

}

import { HashRouter, Routes, Route, Navigate } from 'react-router-dom';
import Layout from './components/Layout';
import Generate from './pages/Generate';
import History from './pages/History';
import Settings from './pages/Settings';
import Home from './pages/Home';
import Queue from './pages/Queue';
import GadgetBlogger from './pages/GadgetBlogger';
import TestPage from './pages/TestPage';
import Dashboard2 from './pages/Dashboard2';
import Blog from './pages/Blog';
import Post from './pages/Post';
import Contact from './pages/Contact';
import Terms from './pages/Terms';
import Privacy from './pages/Privacy';
import { SettingsProvider } from './state/SettingsContext';
import { DraftsProvider } from './state/DraftsContext';
import { useAuthState } from './lib/auth';

function MainPage() {
  const auth = useAuthState();
  
  // 로그인된 사용자는 대시보드로, 아니면 랜딩페이지로
  if (auth.isAuthenticated) {
    return <Navigate to="/app" replace />;
  }
  
  return <Home />;
}

export default function App() {
  return (
    <SettingsProvider>
      <DraftsProvider>
        <HashRouter>
          <Routes>
            <Route path="/" element={<MainPage />} />
            <Route path="/blog" element={<Blog />} />
            <Route path="/post/:id" element={<Post />} />
            <Route path="/contact" element={<Contact />} />
            <Route path="/terms" element={<Terms />} />
            <Route path="/privacy" element={<Privacy />} />
            <Route path="/test" element={<TestPage />} />
            <Route path="/gadget" element={<GadgetBlogger />} />
            <Route path="/dashboard" element={<Dashboard2 />} />
            <Route path="/app" element={<Layout />}> 
              <Route index element={<Dashboard2 />} />
              <Route path="generate" element={<Generate />} />
              <Route path="history" element={<History />} />
              <Route path="queue" element={<Queue />} />
              <Route path="settings" element={<Settings />} />
            </Route>
          </Routes>
        </HashRouter>
      </DraftsProvider>
    </SettingsProvider>
  );
}
